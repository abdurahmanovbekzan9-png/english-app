const SS = SpreadsheetApp.getActiveSpreadsheet(); // Запустите эту функцию вручную (один раз и после каждого обновления скрипта) function setup() { let w = SS.getSheetByName("Words"); if (!w) { w = SS.insertSheet("Words"); w.appendRow(["ID ученика", "Слово", "Перевод", "Уровень", "Верно", "Ошибок", "Добавлено"]); } let p = SS.getSheetByName("Profiles"); if (!p) { p = SS.insertSheet("Profiles"); p.appendRow(["ID ученика", "Имя", "Username", "Уровень", "Цель", "Дата регистрации"]); p.getRange("A1:F1").setFontWeight("bold"); } let s = SS.getSheetByName("Students"); if (!s) s = SS.insertSheet("Students"); s.clear(); s.getRange("A1").setFormula( '=QUERY(Results!A2:J,"select B, max(C), count(I), sum(I) where B is not null group by B label B \'ID ученика\', max(C) \'Имя\', count(I) \'Решено\', sum(I) \'Верно\'",0)' ); s.getRange("E1").setValue("Точность %"); s.getRange("E2").setFormula('=ARRAYFORMULA(IF(C2:C="",,ROUND(D2:D/C2:C*100)))'); s.getRange("A1:E1").setFontWeight("bold"); } function doPost(e) { const d = JSON.parse(e.postData.contents); const lock = LockService.getScriptLock(); lock.waitLock(20000); try { if (d.action === "words") { saveWords(d); } else if (d.action === "profile") { saveProfile(d); } else { SS.getSheetByName("Results").appendRow([ new Date(), d.uid, d.name, d.username, d.sid, d.tense, d.type, d.lvl, d.ok, d.answer ]); } } finally { lock.releaseLock(); } return ContentService.createTextOutput("ok"); } function saveProfile(d) { const sh = SS.getSheetByName("Profiles"); const rows = sh.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(d.uid)) { sh.getRange(i + 1, 1, 1, 6).setValues([[d.uid, d.name, d.username, d.lvl, d.goal, rows[i][5]]]); return; } } sh.appendRow([d.uid, d.name, d.username, d.lvl, d.goal, new Date()]); } function saveWords(d) { const sh = SS.getSheetByName("Words"); const all = sh.getDataRange().getValues(); const head = all[0]; const rest = all.slice(1).filter(r => String(r[0]) !== String(d.uid)); const mine = (d.words || []).map(x => [d.uid, x.w, x.t, x.box || 0, x.c || 0, x.x || 0, x.added || ""]); const out = [head].concat(rest, mine); sh.clearContents(); sh.getRange(1, 1, out.length, 7).setValues(out); } function doGet(e) { const p = e.parameter || {}; const uid = String(p.uid || ""); let out = {}; if (p.action === "words" && uid) { const rows = SS.getSheetByName("Words").getDataRange().getValues().slice(1); out.words = rows.filter(r => String(r[0]) === uid) .map(r => ({ w: r[1], t: r[2], box: r[3], c: r[4], x: r[5], added: r[6] })); } if (p.action === "profile" && uid) { const rows = SS.getSheetByName("Profiles").getDataRange().getValues().slice(1); const r = rows.find(x => String(x[0]) === uid); if (r) out.profile = { name: r[1], lvl: r[3], goal: r[4] }; } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); }
